cmake_minimum_required(VERSION 3.20)
project(GridTalk VERSION 0.4.3 LANGUAGES C)
if(NOT WIN32)
  message(FATAL_ERROR "This plugin targets Windows TeamSpeak")
endif()
option(GRIDTALK_BUILD_TESTS "Build callback test sender" OFF)
if(GRIDTALK_BUILD_TESTS)
  add_executable(test_sender tests/sender.c)
  target_include_directories(test_sender PRIVATE sdk/include)
  target_link_libraries(test_sender PRIVATE assetto_corsa_notifier)
  add_executable(test_lifecycle tests/sender.c)
  target_include_directories(test_lifecycle PRIVATE sdk/include)
  target_compile_definitions(test_lifecycle PRIVATE GRIDTALK_DYNAMIC_LIBRARY=1)
  target_link_libraries(test_lifecycle PRIVATE user32)
  add_executable(test_format tests/formatting.c)
  target_include_directories(test_format PRIVATE sdk/include)
  target_compile_definitions(test_format PRIVATE GRIDTALK_UDP_PORT=19999 GRIDTALK_VERSION="${PROJECT_VERSION}")
  target_link_libraries(test_format PRIVATE ws2_32 user32)
endif()
set(GRIDTALK_UDP_PORT 9999 CACHE STRING "Loopback UDP destination port")
if(NOT GRIDTALK_UDP_PORT MATCHES "^[0-9]+$" OR GRIDTALK_UDP_PORT LESS 1 OR GRIDTALK_UDP_PORT GREATER 65535)
  message(FATAL_ERROR "GRIDTALK_UDP_PORT must be an integer from 1 to 65535")
endif()
add_library(assetto_corsa_notifier SHARED src/plugin.c)
target_include_directories(assetto_corsa_notifier PRIVATE sdk/include)
target_compile_definitions(assetto_corsa_notifier PRIVATE GRIDTALK_UDP_PORT=${GRIDTALK_UDP_PORT} GRIDTALK_VERSION="${PROJECT_VERSION}")
target_link_libraries(assetto_corsa_notifier PRIVATE ws2_32 user32)
set_target_properties(assetto_corsa_notifier PROPERTIES PREFIX "")
if(MSVC)
  target_compile_options(assetto_corsa_notifier PRIVATE /W4)
endif()
