cmake_minimum_required(VERSION 3.20)
project(ts_ac_udp VERSION 0.3.0 LANGUAGES C)
if(NOT WIN32)
  message(FATAL_ERROR "This plugin targets Windows TeamSpeak")
endif()
option(TS_AC_BUILD_TESTS "Build callback test sender" OFF)
if(TS_AC_BUILD_TESTS)
  add_executable(test_sender tests/sender.c)
  target_include_directories(test_sender PRIVATE sdk/include)
  target_link_libraries(test_sender PRIVATE ts_ac_udp)
endif()
set(TS_AC_UDP_PORT 9999 CACHE STRING "Loopback UDP destination port")
add_library(ts_ac_udp SHARED src/plugin.c)
target_include_directories(ts_ac_udp PRIVATE sdk/include)
target_compile_definitions(ts_ac_udp PRIVATE TS_AC_UDP_PORT=${TS_AC_UDP_PORT})
target_link_libraries(ts_ac_udp PRIVATE ws2_32 user32)
set_target_properties(ts_ac_udp PROPERTIES PREFIX "")
if(MSVC)
  target_compile_options(ts_ac_udp PRIVATE /W4)
endif()
